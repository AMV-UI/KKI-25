# Check if we are in HOLD mode condition
class CheckPXMode(py_trees.behaviour.Behaviour):
    def __init__(self):
        super(CheckPXMode, self).__init__(name="Check PX Mode")
        self.blackboard = py_trees.blackboard.Client(name="CheckPXMode")
        self.blackboard.register_key("pxmode", access=py_trees.common.Access.READ)
        
    def update(self):
        if self.blackboard.pxmode == PxMode.HOLD.value:
            return py_trees.common.Status.SUCCESS
        else:
            return py_trees.common.Status.FAILURE

# Condition: Check if detection is stable for some frames
class IsDetectionStable(py_trees.behaviour.Behaviour):
    def __init__(self):
        super(IsDetectionStable, self).__init__(name="Detection Stable")
        self.blackboard = py_trees.blackboard.Client(name="IsDetectionStable")
        self.blackboard.register_key("detected", access=py_trees.common.Access.READ)
        self.blackboard.register_key("frame_counter", access=py_trees.common.Access.WRITE)
        
    def update(self):
        if self.blackboard.detected:
            self.blackboard.frame_counter.is_started()
            if self.blackboard.frame_counter.is_enough():
                self.blackboard.frame_counter.reset()
                return py_trees.common.Status.SUCCESS
        else:
            self.blackboard.frame_counter.reset()
        return py_trees.common.Status.FAILURE

# Action: Toggle Track

    def update(self):
        current_track = self.blackboard.find_mode.get_track()
        new_track = "B" if current_track == "A" else "A"
        self.blackboard.find_mode.set_track(new_track)
        self.node.get_logger().info(f"Switching track from {current_track} to {new_track}")
        return py_trees.common.Status.SUCCESS

# Action: Approach Buoy - drive forward when buoy is detected
class ApproachBuoy(py_trees.behaviour.Behaviour):
    def __init__(self, node):
        super(ApproachBuoy, self).__init__(name="Approach Buoy")
        self.node = node
        self.blackboard = py_trees.blackboard.Client(name="ApproachBuoy")
        
        # Publishers
        self.state_dst_pub = node.create_publisher(
            Float64,
            Topic.state_dst,
            10
        )
        self.mission_pub = node.create_publisher(
            UInt8,
            Topic.mission,
            10
        )
        
    def initialise(self):
        self.node.get_logger().info(f"<=> [{NodeConfig.mission}] Approaching buoy")
        
    def update(self):
        # Default center position
        dsc_state = 160.0  
        self.state_dst_pub.publish(Float64(data=dsc_state))
        
        # Publish current mission (just for logging/status purposes)
        self.mission_pub.publish(UInt8(data=1))
        
        return py_trees.common.Status.RUNNING

# Action: Search for buoy using find_mode
class SearchForBuoy(py_trees.behaviour.Behaviour):
    def __init__(self, node):
        super(SearchForBuoy, self).__init__(name="Search For Buoy")
        self.node = node
        self.blackboard = py_trees.blackboard.Client(name="SearchForBuoy")
        self.blackboard.register_key("px_heading", access=py_trees.common.Access.READ)
        self.blackboard.register_key("find_mode", access=py_trees.common.Access.READ)
        
        # Publishers
        self.state_dst_pub = node.create_publisher(
            Float64,
            Topic.state_dst,
            10
        )
        self.mission_pub = node.create_publisher(
            UInt8,
            Topic.mission,
            10
        )
        
    def initialise(self):
        self.node.get_logger().info(f"<=> [{NodeConfig.mission}] Searching for buoy")
        
    def update(self):
        # Use find_mode to determine steering direction
        if self.blackboard.px_heading >= 0:
            heading_state = self.blackboard.find_mode.get_state(self.blackboard.px_heading)
            self.node.get_logger().info(f"[{NodeConfig.mission}] Current Heading: {self.blackboard.px_heading}, Command: {heading_state}")
        else:
            heading_state = 160.0  # Default center if no heading
            
        # Publish steering command
        self.state_dst_pub.publish(Float64(data=heading_state))
        
        # Publish current mission (just for logging/status purposes)
        self.mission_pub.publish(UInt8(data=0))
        
        return py_trees.common.Status.RUNNING

# Action: Perform maneuvering after buoy detection
class PerformManeuver(py_trees.behaviour.Behaviour):
    def __init__(self, node):
        super(PerformManeuver, self).__init__(name="Perform Maneuver")
        self.node = node
        self.blackboard = py_trees.blackboard.Client(name="PerformManeuver")
        self.blackboard.register_key("detected", access=py_trees.common.Access.READ)
        self.blackboard.register_key("manuver_detected", access=py_trees.common.Access.WRITE)
        self.blackboard.register_key("find_mode", access=py_trees.common.Access.READ)
        self.blackboard.register_key("frame_counter_manuver", access=py_trees.common.Access.WRITE)
        
        # Publishers
        self.state_dst_pub = node.create_publisher(
            Float64,
            Topic.state_dst,
            10
        )
        self.mission_pub = node.create_publisher(
            UInt8,
            Topic.mission,
            10
        )
        
    def initialise(self):
        self.node.get_logger().info(f"<=> [{NodeConfig.mission}] Performing maneuver")
        
    def update(self):
        # Set mission number for logging
        self.mission_pub.publish(UInt8(data=6))
        
        if self.blackboard.detected:
            self.blackboard.manuver_detected = True
            # Maneuver direction
            if self.blackboard.find_mode.get_track() == "A":
                # Left
                self.state_dst_pub.publish(Float64(data=-200))
                self.node.get_logger().info("Maneuvering left")
            else:
                # Right
                self.state_dst_pub.publish(Float64(data=200))
                self.node.get_logger().info("Maneuvering right")
        elif self.blackboard.manuver_detected == True and not self.blackboard.detected:
            self.blackboard.frame_counter_manuver.is_started()
            # Continue maneuver
            if self.blackboard.find_mode.get_track() == "A":
                self.state_dst_pub.publish(Float64(data=-200))
            else:
                self.state_dst_pub.publish(Float64(data=200))

            if self.blackboard.frame_counter_manuver.is_enough():
                return py_trees.common.Status.SUCCESS
        else:
            # Default behavior when nothing is detected yet
            self.state_dst_pub.publish(Float64(data=160.0))

        return py_trees.common.Status.RUNNING
